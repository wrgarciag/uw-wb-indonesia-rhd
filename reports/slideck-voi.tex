\begin{frame}{We value perfect knowledge of four prophylaxis inputs for
one decision}
\phantomsection\label{we-value-perfect-knowledge-of-four-prophylaxis-inputs-for-one-decision}
\begin{columns}[T]
\begin{column}{0.44\linewidth}
\textbf{One decision, 2026--2050}

Current care, or scale up echo screening plus secondary antibiotic
prophylaxis (SAP)?
\end{column}

\begin{column}{0.52\linewidth}
\textbf{Four uncertain SAP inputs}

\begin{enumerate}
\tightlist
\item
  Efficacy, stage A \(\rightarrow\) B (minimal \(\rightarrow\) mild)
\item
  Efficacy, stage B \(\rightarrow\) C (mild \(\rightarrow\) advanced)
\item
  Age to which protection lasts
\item
  Protection with partial adherence
\end{enumerate}
\end{column}
\end{columns}

\medskip

\begin{uncertaintyblock}{Held fixed}
Disease model, care cascade, costs, valuation. No SAP effect on late stages or on deaths.
\end{uncertaintyblock}
\end{frame}

\bignumber{0.09}{risk ratio for progression of early RHD on prophylaxis (95\% CI 0.03--0.29)}

\medskip

\begin{frame}{Information values come from one set of 500 paired draws
per country}
\phantomsection\label{information-values-come-from-one-set-of-500-paired-draws-per-country}
\[\mathrm{EVPI} = \mathrm{E}_\theta\big[\max_d \mathrm{NB}(d,\theta)\big] - \max_d \mathrm{E}_\theta\big[\mathrm{NB}(d,\theta)\big]\]

\begin{itemize}
\tightlist
\item
  Joint value of the four inputs = EVPI, only because nothing else
  varies here
\item
  Single input: GAM regression of B and C on that input
  \parencite{Strong2014}
\item
  Single-input values overlap; never summed or stacked
  \parencite{Rothery2020}
\item
  Intervals: 1,000 bootstrap resamples of whole paired draws (estimator
  precision)
\end{itemize}
\end{frame}
